\section{Requisitos}
\label{sec:3.1}
% Requisitos del sistema

\begin{enumerate}
	\item \textbf{Requisitos Funcionales}
	\begin{itemize}
		\item RF 1: Se puede podrá escribir en el teclado
		\item RF 2: Habrá un teclado virtual que simula uno físico y un teclado plano. 
		\item RF 3: Se puede cambiar entre el teclado físico y el plano cuando se requiera. 
		\item RF 4: Los teclados tendrán una versión funcional en 8 idiomas establecidos.
		\item RF 5: Se puede cambiar de idioma cuando se requiera. 
		\item RF 6: Escritura por voz funcional para todos los idiomas descritos y en los dos tipos de teclado.
		\item RF 7: Escritura tipo swipe funcional para todos los idiomas descritos en el teclado plano solo.
	\end{itemize}
	\item \textbf{Requisitos No Funcionales}
	\begin{itemize}
		\item RNF 1: El sistema debe ser desarrollado usando Unity. 
		\item RNF 2: Lenguaje de desarrollo C\#. 
	\end{itemize}
\end{enumerate}





\section{Especificaciones}
% Especificación del sistema a partir de lo recogido en los requisitos
A continuación voy a redactar las especificaciones de los requisitos siguiendo el estándar IEEE \cite{ieee830-1993}


\begin{enumerate}
	\item \textbf{Introducción}
	\begin{enumerate}
		\item \textbf{Propósito:} El propósito de esta especificación es definir los requisitos del sistema para el desarrollo del teclado virtual multilingüe en Unity, proporcionando una guía clara y detallada para los desarrolladores y las partes interesadas.
		\item \textbf{Alcance:} Esta especificación cubre los requisitos funcionales y no funcionales del sistema, así como las interfaces de usuario, limitaciones conocidas y futuras extensiones posibles.
		\item \textbf{Definiciones, Acrónimos y Abreviaturas:}
		\begin{itemize}
			\item \textbf{RF}: Requisito Funcional
			\item \textbf{RNF}: Requisito No Funcional
			\item \textbf{VR}: Realidad Virtual
			\item \textbf{SDK}: Kit de Desarrollo de Software
		\end{itemize}
	\end{enumerate}
	\item \textbf{Descripción General}
	\begin{enumerate}
		\item \textbf{Perspectiva del Producto:} El sistema consistirá en un teclado virtual que simula uno físico y uno plano en un entorno de realidad virtual (VR). Los usuarios podrán cambiar entre estos teclados y seleccionar el idioma deseado para la entrada de texto.
		\item \textbf{Funciones del Producto:} El teclado virtual permitirá:
		\begin{itemize}
			\item Cambiar entre teclado físico y plano.
			\item Seleccionar entre seis idiomas disponibles.
		\end{itemize}
		\item \textbf{Características del Usuario:} Los usuarios serán principalmente aquellos que utilicen aplicaciones de realidad virtual donde se requiera la entrada de texto, como aplicaciones educativas y de entretenimiento.
	\end{enumerate}
	\item \textbf{Requisitos Específicos}
	\begin{enumerate}
		\item \textbf{Requisitos Funcionales}
		\begin{enumerate}
			\item \textbf{RF1: Cambio entre Teclado Físico y Teclado Plano:} El sistema debe permitir al usuario cambiar entre un teclado virtual que simula uno físico y uno plano.
			\item \textbf{RF2: Selección de Idioma:} El sistema debe permitir al usuario seleccionar entre seis idiomas disponibles para el teclado virtual.
		\end{enumerate}
		
		\item \textbf{Requisitos No Funcionales}
		\begin{enumerate}
			\item \textbf{RNF1: Plataforma de Desarrollo:} El sistema debe ser desarrollado utilizando Unity como plataforma principal.
			\item \textbf{RNF2: Lenguaje de Desarrollo:} El sistema debe ser desarrollado utilizando el lenguaje de programación C\#.
		\end{enumerate}
	\end{enumerate}
	\item \textbf{Otros Requisitos}
	\begin{enumerate}
		\item \textbf{Interfaces de Usuario:} El sistema deberá incluir interfaces de usuario intuitivas y fáciles de usar para la selección de teclado, idioma y entrada de texto.
	\end{enumerate}

\section{Viabilidad}
% Viabilidad del proyecto presentado

\subsection{Viabilidad Técnica}
\begin{itemize}

	\item Plataforma de Desarrollo:
	\begin{itemize}
		\item El uso de Unity como plataforma desarroladora, ofrece una gran robustez a la hora de desarollar aplicaciones y Assets de realidad virtual, ya que es una plataformal bien establecida en este ámbito.
		\item Al Unity ofrecer un soporte multiplataforma, permite el desarrollo para distintos dispositivos de realidad virtual.
		\item Yo poseo ya suficiente experiencia en el uso de esta plataforma.
		
	\end{itemize}
	\item Lenguaje de Programación:
	\begin{itemize}
		\item Se utilizará C\# como lenguaje de desarrollo tanto por ser el lenguaje principal de Unity, por contar con una gran cantidad de recursos y comunidades de soporte y por su amplia utilización de mi parte durante el transcurso del grado universitario.
	\end{itemize}
	
	\item Bibliotecas y Herramientas Adicionales:
	\begin{itemize}
		\item XR Interaction Toolkit: para la implementación de las funcionalidades en VR del proyecto.
		\item OpenXR Pluguin: pluguin usado para la compatibilidad con múltiples dispositivos VR.
		\item XR Hands: para permitir el funcionamiento del hand tracking en dispositivos habilitados para ello.
		\item TextMesh Pro: usado para poder formatear el texto y que permita el uso de caracteres UNICODE en el interfaz de usuario.
	\end{itemize}
	
	\item Pruebas y Depuración:
	\begin{itemize}
		\item Al Unity proporcionar un entorno integrado para pruebas y depuración, facilita la identificación y resolución de errores durante el desarrollo.
		\item Tengo el permiso de probar la aplicación en una clase del \textit{FP Superior de Técnico en Desarrollo de Aplicaciones Multiplataforma} en el instituti IES Jaume II el Just de Tavernes de la Valldigna.
	\end{itemize}
\end{itemize}
