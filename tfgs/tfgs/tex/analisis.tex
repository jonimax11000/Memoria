\section{Análisis de comportamiento}
%Escenarios de Uso:
%Descripción detallada de escenarios típicos de uso.
%Ejemplos prácticos de cómo se utilizará el sistema.

\begin{itemize}
	\item Descripción detallada de escenarios típicos de uso:
	\begin{itemize}
		\item Escenario 1: Cambio de idioma en el teclado virtual
		\begin{itemize}
			\item El usuario inicia la aplicación de realidad virtual.
			\item El teclado aparece con el idioma del sistema operativo si coincide con uno de los idiomas disponibles, si no, aparece en inglés.
			\item El usuario selecciona el menú de configuración del teclado.
			\item El usuario elige uno de los seis idiomas disponibles.
			\item El teclado virtual cambia al idioma seleccionado.
		\end{itemize}
		\item Escenario 2: Cambio de tipo de teclado (físico y plano)
		\begin{itemize}
			\item El usuario inicia la aplicación de realidad virtual.
			\item El teclado virtual aparece en su configuración predeterminada, que es el teclado físico.
			\item El usuario pulsa el botón de cambio de teclado.
			\item El teclado virtual cambia de una representación física a una plana.
		\end{itemize}
	\end{itemize}
	
	\item Ejemplos prácticos de cómo se utilizará el sistema:
	\begin{itemize}
		\item Ejemplo 1: Uso en un juego de RV en varios idiomas
		\begin{itemize}
			\item El usuario crea el juego y necesita usar el Asset.
			\item El usuario selecciona el tipo de teclado (físico o plano) según su preferencia.
			\item El usuario cambia la forma de cambiar de idioma para adaptarlo a su propio menú.
			\item El usuario configura el juego y el teclado según sus necesidades.
		\end{itemize}
		
		\item Ejemplo 2: Redacción de datos en un programa de RV
		\begin{itemize}
			\item El usuario crea el programa y se descarga el Asset.
			\item El usuario pone el teclado virtual en el idioma deseado.
			\item El usuario selecciona el tipo de teclado (físico o plano) según su preferencia.
			\item El usuario configura la introducción de datos para usar el teclado.
		\end{itemize}
		
		\item Ejemplo 3: Simulación de mecanografía en RV
		\begin{itemize}
			\item El usuario crea el simulador y se descarga el Asset.
			\item El usuario pone el teclado virtual en el idioma deseado.
			\item El usuario selecciona el de teclado físico o plano.
			\item El usuario configura la introducción de datos para usar el teclado.
			\item El usuario configura los objetivos mecanográficos necesarios.
		\end{itemize}
	\end{itemize}
\end{itemize}
